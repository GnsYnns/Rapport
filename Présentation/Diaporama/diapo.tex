\section{3. Numérique responsable}

\begin{frame}{3. Numérique responsable}{3. Numérique responsable -- 3\,min}
\begin{block}{Question centrale}
L'IT de Haute-Goulaine est-elle alignée sur la stratégie RSE du groupe ?
\end{block}
\vfill
\begin{itemize}
  \item Point d'entrée : poste de l'apprenti, pile Windows/WSL, CI/CD
  \item C'est la partie la plus personnelle et la plus opérationnelle
\end{itemize}
\end{frame}

\begin{frame}{Constat : des atouts, pas encore un programme}{3. Numérique responsable -- 3\,min}
\begin{columns}[T]
  \begin{column}{0.55\textwidth}
    \begin{itemize}
      \item \textbf{Privacy by Design} intégré ; \textbf{PIA} à systématiser
      \item Briques numériques solides : Web Academy, IMDS, ESGeo
      \item Atouts locaux : virtualisation WSL, CI/CD, revues de code
      \item \alert{\textbf{Mais} : aucun programme \textit{Green IT} interne documenté}
    \end{itemize}
  \end{column}
  \begin{column}{0.42\textwidth}
    \begin{alertblock}{Message clé}
      Le constat central : la direction IT groupe ne documente pas encore de politique Green IT appliquée aux postes, serveurs et cycle de vie du matériel.
    \end{alertblock}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Où en sont les concurrents ?}{3. Numérique responsable -- 3\,min}
\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \begin{block}{Volvo Group}
      Programme \textit{Green by Design} : cloud, matériel circulaire, empreinte numérique des salariés
    \end{block}
    \begin{block}{Scania}
      Mesure de l'impact applications (\textit{vCPU.h}), réemploi du matériel
    \end{block}
  \end{column}
  \begin{column}{0.48\textwidth}
    \begin{block}{Daimler Truck}
      \textbf{-43\,\%} d'émissions Scope 1 \& 2 depuis 2021
    \end{block}
    \begin{alertblock}{Positionnement Iveco}
      Briques solides mais pas encore de programme Green IT interne publié
    \end{alertblock}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{3 recommandations phares pour Haute-Goulaine}{3. Numérique responsable -- 3\,min}
\begin{enumerate}
  \item \textbf{Éco-conception logicielle} : checklist Green IT, optimisation des requêtes, extinction automatique des environnements non productifs
  \item \textbf{Cycle de vie du matériel} : atteindre \textbf{30\,\%} de parc test/reconditionné d'ici 2028
  \item \textbf{Achats responsables} : \textbf{100\,\%} des nouveaux postes labellisés d'ici 2026
\end{enumerate}
\vspace{0.3cm}
\begin{block}{KPI transverse}
  Réduire de \textbf{15\,\%} la consommation moyenne CPU/mémoire des serveurs de développement d'ici fin 2027.
\end{block}
\end{frame}
